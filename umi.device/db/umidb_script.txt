CREATE TABLE situacao(ST_CONTRIBUINTE TEXT, SITUACAO TEXT); 
CREATE INDEX situacaoST ON situacao (ST_CONTRIBUINTE);
CREATE TABLE contribuinte(SQ_PESSOA NUMERIC, SQ_CONTRIBUINTE NUMERIC, INSCRICAO_ESTADUAL TEXT, NM_FORMAL TEXT, NU_CNPJ TEXT, ST_CONTRIBUINTE TEXT, DT_ULTIMA_ATUALIZACAO DATETIME, FOREIGN KEY (ST_CONTRIBUINTE) REFERENCES situacao(ST_CONTRIBUINTE), PRIMARY KEY (SQ_PESSOA, SQ_CONTRIBUINTE));
CREATE INDEX contribuinteIE ON contribuinte (INSCRICAO_ESTADUAL);
CREATE INDEX contribuinteSQ ON contribuinte (SQ_PESSOA, SQ_CONTRIBUINTE);
CREATE VIEW vwcontribuinte AS
SELECT contribuinte.SQ_PESSOA AS SQ_PESSOA, contribuinte.SQ_CONTRIBUINTE AS SQ_CONTRIBUINTE, contribuinte.INSCRICAO_ESTADUAL AS INSCRICAO_ESTADUAL, 
contribuinte.NM_FORMAL AS NM_FORMAL, contribuinte.NU_CNPJ AS NU_CNPJ, contribuinte.ST_CONTRIBUINTE AS ST_CONTRIBUINTE, situacao.SITUACAO AS SITUACAO, 
contribuinte.DT_ULTIMA_ATUALIZACAO AS DT_ULTIMA_ATUALIZACAO
FROM contribuinte INNER JOIN situacao ON contribuinte.ST_CONTRIBUINTE = situacao.ST_CONTRIBUINTE;
CREATE TABLE usuario(SQ_PESSOA NUMERIC, NM_FORMAL TEXT, CD_USUARIO TEXT, PASSWORD TEXT, DT_ULTIMO_LOGIN DATETIME, PRIMARY KEY (SQ_PESSOA));
CREATE INDEX usuarioCD ON usuario (CD_USUARIO);
